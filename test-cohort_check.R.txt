# ============================================================================
# test-cohort_check.R -- automated tests for the cohort check
# ============================================================================
# Run with:   testthat::test_file("test-cohort_check.R")
# (this file must sit in the same folder as cohort_check.R)
#
# Each test states, in its name, the plain-English fact it verifies.
# If a test fails after you change the code, the code broke that fact.
# ============================================================================

library(testthat)
source("cohort_check.R")


test_that("allocating exports to birth cohorts never loses or invents animals", {
  # Whatever the spreading arithmetic does, the total number of animals
  # allocated to cohorts must equal the total number exported.
  fake <- make_fake_data(constant = TRUE)
  allocated <- allocate_exports_to_cohorts(fake$exports)
  expect_equal(sum(allocated$removed), sum(fake$exports$n), tolerance = 1e-9)
})


test_that("net calf crop equals births when nothing leaves the cohort", {
  # With zero exports and zero calf slaughterings, ncc must equal births.
  months_all <- seq(as.Date("2020-01-01"), by = "month", length.out = 24)
  births  <- tibble(month = months_all, births = 1000)
  exports <- tibble(month = months_all, age_band = "0-3", n = 0)
  kill    <- tibble(month = months_all, steers = 0, heifers = 0,
                    bulls = 0, cows = 0, calves = 0)
  ncc <- net_calf_crop(births, exports, kill)
  expect_equal(ncc$ncc, ncc$births)
})


test_that("the ratio recovers the known constant built into the fake data", {
  # make_fake_data(constant = TRUE) constructs the prime kill as exactly
  # 0.6 x supply, so the check must report a ratio of exactly 0.6.
  fake <- make_fake_data(constant = TRUE)
  res  <- cohort_check(fake$births, fake$exports, fake$kill)
  latest_ratio <- tail(na.omit(res$ratio), 1)
  expect_equal(latest_ratio, 0.6, tolerance = 1e-6)
})


test_that("clean data passes the flag check", {
  fake <- make_fake_data(constant = TRUE)
  res  <- cohort_check(fake$births, fake$exports, fake$kill)
  expect_equal(flag_latest_month(res)$result, "Pass")
})


test_that("a large error in the latest month is flagged", {
  # Halve the latest month's prime kill -- as if a big plant's return went
  # missing -- and the verdict must flip to FLAG.
  fake <- make_fake_data(constant = TRUE)
  kill_broken <- fake$kill
  last_row <- which.max(kill_broken$month)
  kill_broken[last_row, c("steers", "heifers", "bulls")] <-
    kill_broken[last_row, c("steers", "heifers", "bulls")] * 0.5

  res <- cohort_check(fake$births, fake$exports, kill_broken)
  expect_equal(flag_latest_month(res)$result, "FLAG")
})


test_that("a helpful error is raised when an input column is missing", {
  fake <- make_fake_data(constant = TRUE)
  births_bad <- rename(fake$births, calf_births = births)  # wrong column name
  expect_error(
    cohort_check(births_bad, fake$exports, fake$kill),
    "missing column"
  )
})


test_that("a helpful error is raised for an unknown export age band", {
  fake <- make_fake_data(constant = TRUE)
  exports_bad <- fake$exports
  exports_bad$age_band[1] <- "36-48"   # a band we have no definition for
  expect_error(
    cohort_check(fake$births, exports_bad, fake$kill),
    "don't recognise"
  )
})


test_that("a helpful error is raised when a month is missing from the sequence", {
  fake <- make_fake_data(constant = TRUE)
  births_gappy <- filter(fake$births, month != as.Date("2020-06-01"))
  expect_error(
    cohort_check(births_gappy, fake$exports, fake$kill),
    "missing month"
  )
})
